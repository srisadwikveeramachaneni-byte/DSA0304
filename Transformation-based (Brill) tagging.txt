
def initial_tag(word):
    if word in ["the", "a", "an"]:
        return "DT"
    elif word.endswith("ing"):
        return "VBG"
    elif word.endswith("ly"):
        return "RB"
    else:
        return "NN"

def transformation_tag(words):
    tagged = []

    for word in words:
        tag = initial_tag(word)
        tagged.append([word, tag])

    # Transformation rule:
    # Change NN to VB when the word follows "to".
    for i in range(1, len(tagged)):
        if tagged[i-1][0] == "to":
            if tagged[i][1] == "NN":
                tagged[i][1] = "VB"

    return tagged

sentence = ["the", "students", "to", "dance", "quickly"]

result = transformation_tag(sentence)

print("Input:", sentence)
print("Transformed Tags:")

for item in result:
    print(item)
