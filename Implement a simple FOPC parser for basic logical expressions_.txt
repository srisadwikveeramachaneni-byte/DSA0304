
import re

def parse_fopc(expression):
    tokens = re.findall(
        r'forall|exists|[A-Za-z_]\w*|->|[()&,|]',
        expression
    )

    print("Tokens:", tokens)

    if not tokens:
        return False

    balance = 0

    for token in tokens:
        if token == "(":
            balance += 1
        elif token == ")":
            balance -= 1

        if balance < 0:
            return False

    if balance != 0:
        return False

    return True

expression = "forall x (Human(x) -> Mortal(x))"

print("Expression:", expression)

if parse_fopc(expression):
    print("Valid basic logical expression")
else:
    print("Invalid expression")
