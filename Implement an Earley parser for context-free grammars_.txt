
grammar = {
    "S": [["NP", "VP"]],
    "NP": [["Det", "N"]],
    "VP": [["V", "NP"]],
    "Det": [["the"], ["a"]],
    "N": [["cat"], ["dog"]],
    "V": [["sees"], ["likes"]]
}

def earley_parser(words):
    chart = [set() for _ in range(len(words) + 1)]

    chart[0].add(("S", ("NP", "VP"), 0, 0))

    for i in range(len(words) + 1):
        changed = True

        while changed:
            changed = False

            for state in list(chart[i]):
                lhs, rhs, dot, start = state

                if dot < len(rhs):
                    symbol = rhs[dot]

                    if symbol in grammar:
                        for production in grammar[symbol]:
                            new_state = (
                                symbol, tuple(production), 0, i
                            )

                            if new_state not in chart[i]:
                                chart[i].add(new_state)
                                changed = True

                    elif i < len(words) and symbol == words[i]:
                        chart[i + 1].add(
                            (lhs, rhs, dot + 1, start)
                        )

                else:
                    for prev in list(chart[start]):
                        plhs, prhs, pdot, pstart = prev

                        if pdot < len(prhs) and prhs[pdot] == lhs:
                            new_state = (
                                plhs, prhs, pdot + 1, pstart
                            )

                            if new_state not in chart[i]:
                                chart[i].add(new_state)
                                changed = True

    return (
        "S", ("NP", "VP"), 2, 0
    ) in chart[len(words)]

sentence = "the cat sees a dog".split()

if earley_parser(sentence):
    print("Sentence accepted by Earley Parser")
else:
    print("Sentence rejected by Earley Parser")
